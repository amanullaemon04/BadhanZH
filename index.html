function populateCities(donors) { if (!cityEl) return; // Keep the 64 Bangladesh districts already defined in index.html. // Do not replace the dropdown with only cities found in donor records. const current = cityEl.value; // If the HTML already contains the district list, // preserve it exactly as it is. if (cityEl.options.length > 1) { const currentNormalized = normalize(current); const matchingOption = Array.from(cityEl.options).find( (option) => normalize(option.value) === currentNormalized ); cityEl.value = matchingOption ? matchingOption.value : ""; return; } // Fallback only if the HTML dropdown has no district options. const cities = [ ...new Set( donors .map((x) => String(x.city ?? "").trim()) .filter(Boolean) ) ].sort((a, b) => a.localeCompare(b)); cityEl.innerHTML = ""; const all = document.createElement("option"); all.value = ""; all.textContent = "All districts"; cityEl.appendChild(all); cities.forEach((city) => { const option = document.createElement("option"); option.value = city; option.textContent = city; cityEl.appendChild(option); }); }
